        Args:
            target_density_matrix (np.ndarray): The desired 2x2 density matrix representing

        Returns:
            float: The calculated RCF value, ranging from 0.0 to 1.0.

        Raises:
            ValueError: If the target_density_matrix is not a valid 2x2 matrix or not Hermitian.
        """

            logging.error("Invalid target_density_matrix: Must be a 2x2 numpy array.")
            raise ValueError("target_density_matrix must be a 2x2 numpy array.")
        if not np.allclose(target_density_matrix, target_density_matrix.conj().T):
            logging.error("Invalid target_density_matrix: Must be Hermitian.")
            raise ValueError("target_density_matrix must be Hermitian.")
        if not np.isclose(np.trace(target_density_matrix), 1.0):
            logging.warning("Target density matrix trace is not 1. Normalizing for calculation.")
            target_density_matrix /= np.trace(target_density_matrix)

        for i in range(self._rpu_count):

        with self._lock:
            self._current_rcf = rcf
        logging.debug(f"Calculated RCF: {rcf:.4f}")
        return rcf
